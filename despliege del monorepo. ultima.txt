# ==============================================================================
# SCRIPT DE DESPLIEGUE DIRECTO A PRODUCCIÓN (RAMA MAIN)
# ==============================================================================
$ErrorActionPreference = "Stop"

Set-Location "C:\Users\alfon\Desktop\CLOUD_ISV_DESARROLLO_AGOSTO"

# 1. Limpieza de artefactos
Write-Host "🧹 Limpiando caché e instalando dependencias..." -ForegroundColor Yellow
pnpm run clean
pnpm install --no-frozen-lockfile

# 2. Verificación de Control Plane Local
Write-Host "⚙️ Verificando Control Plane Local..." -ForegroundColor Cyan
if (Test-Path "control-plane-local\package.json") {
    pnpm --filter control-plane-local run build
}

# 3. Puerta de Resiliencia
Write-Host "🛡️️ Ejecutando Puerta de Resiliencia..." -ForegroundColor Cyan
if (Test-Path "resilienceGate_2_2.js") { 
    node resilienceGate_2_2.js 
} elseif (Test-Path "resilienceGate.js") { 
    node resilienceGate.js 
}

# 4. Compilación paralela con Turbo
Write-Host "🏗️ Compilando aplicaciones con Turborepo..." -ForegroundColor Cyan
pnpm run build

# 5. Empaquetado L7
Write-Host "🧩 Empaquetando Sensor L7..." -ForegroundColor Cyan
if (Test-Path "extension-base") {
    if (-not (Test-Path "saare-console\public")) { New-Item -ItemType Directory -Path "saare-console\public" -Force }
    Compress-Archive -Path "extension-base\*" -DestinationPath "saare-console\public\extension-base.zip" -Force
}

# 6. Despliegue en PRODUCCIÓN (--branch=main)
Write-Host "🚀 Desplegando en PRODUCCIÓN de Cloudflare..." -ForegroundColor Cyan
npx wrangler pages deploy saare-console/dist --project-name=saare-console --branch=main --commit-dirty=true
npx wrangler pages deploy saare-suite/dist --project-name=saare-suite --branch=main --commit-dirty=true

if (Test-Path "control-plane") {
    Set-Location control-plane
    npx wrangler deploy
    Set-Location ..
}

# 7. Sincronización en GitHub
Write-Host "📦 Sincronizando repositorio en GitHub..." -ForegroundColor Cyan
git add .
$gitStatus = git status --porcelain
if ($gitStatus) {
    git commit -m "chore(release): despliegue oficial de produccion v8.2 en main"
    git push origin main
    Write-Host "🐙 Cambios subidos a GitHub." -ForegroundColor Green
} else {
    Write-Host "⚠️ No hay cambios pendientes en Git." -ForegroundColor Yellow
}

Write-Host "`n✅ ¡DESPLIEGUE DE PRODUCCIÓN COMPLETADO!" -ForegroundColor Green
Write-Host "  • Landing Comercial: https://saare.es / https://saare-suite.pages.dev" -ForegroundColor Green
Write-Host "  • Consola GRC:       https://consola.saare.es / https://saare-console.pages.dev" -ForegroundColor Green
Write-Host "  • Control Plane API: https://api.saare.es" -ForegroundColor Green
