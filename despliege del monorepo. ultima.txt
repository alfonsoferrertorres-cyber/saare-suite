# 1. Posicionamiento e interrupción en caso de error
$ErrorActionPreference = "Stop"
Set-Location "C:\Users\alfon\Desktop\CLOUD_ISV_DESARROLLO_AGOSTO"

# 2. Instalación de dependencias del monorepo
Write-Host "📦 Instalando dependencias..." -ForegroundColor Cyan
pnpm install --no-frozen-lockfile

# 3. Compilación limpia del monorepo
Write-Host "🏗️ Compilando aplicaciones..." -ForegroundColor Cyan
pnpm run clean
pnpm run build:all

# 4. Despliegue de Control Plane (Workers)
Write-Host "🚀 Desplegando Control Plane..." -ForegroundColor Cyan
Set-Location control-plane
npx wrangler deploy
Set-Location ..

# 5. Empaquetado del Sensor L7 Sovereign
Write-Host "🧩 Empaquetando Sensor L7..." -ForegroundColor Cyan
$publicDir = "saare-console\public"
$zipPath = "$publicDir\extension-base.zip"
if (-not (Test-Path $publicDir)) { New-Item -ItemType Directory -Path $publicDir -Force }
if (Test-Path $zipPath) { Remove-Item -Force $zipPath }
Compress-Archive -Path "extension-base\*" -DestinationPath $zipPath -Force

# 6. Despliegue de Consola GRC y Landing Page (Pages) con comprobación previa
Write-Host "🌐 Desplegando Cloudflare Pages..." -ForegroundColor Cyan
if (Test-Path "saare-console\dist\index.html") {
    npx wrangler pages deploy saare-console/dist --project-name=saare-console --commit-dirty=true
} else { Write-Error "saare-console/dist/index.html no existe."; exit 1 }

if (Test-Path "saare-suite\dist\index.html") {
    npx wrangler pages deploy saare-suite/dist --project-name=saare-suite --commit-dirty=true
} else { Write-Error "saare-suite/dist/index.html no existe."; exit 1 }

# 7. Sincronización en Git y GitHub
Write-Host "📦 Sincronizando repositorio con GitHub..." -ForegroundColor Cyan
git add .
if (git status --porcelain) {
    git commit -m "chore(release): despliegue completo de producción v8.1"
    git push origin main
    Write-Host "🐙 Cambios subidos a GitHub correctamente." -ForegroundColor Green
}

Write-Host "`n✅ ¡Despliegue completado con éxito!" -ForegroundColor Green
